cmake_minimum_required(VERSION 3.22)
project(feberis_core_tests C)
set(CMAKE_C_STANDARD 11)
option(FEB_SANITIZERS "Enable address/undefined behavior sanitizers" ON)
enable_testing()
add_executable(test_core test_core.c ../main/core/nmea.c ../main/core/wire.c
    ../main/core/wigle.c ../main/core/delivery.c ../main/core/wardrive.c)
target_include_directories(test_core PRIVATE ../main/core)
target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(test_core PRIVATE m)
if(FEB_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_core PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME core COMMAND test_core)
